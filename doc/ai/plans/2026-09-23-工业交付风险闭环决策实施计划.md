# NexusAI 工业交付风险闭环决策 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在不破坏现有通用 AI 对话链的前提下，交付“交付风险识别—缺料与 APS 推演—证据化决策—人工审批—安全执行—结果评测”的竞赛版工业决策闭环。

**Architecture:** 共享契约和持久化模型放在 `ruoyi-common-ai`；`ruoyi-erp` 通过 `IndustrialDecisionDataPort` 提供单一业务口径；`ruoyi-ai` 负责评分、证据、摘要、策略、状态机与 API；`ruoyi-flow` 通过 `DecisionWorkflowPort` 编排分析/执行流程。PC 与 App 只消费结构化 `DecisionResult`，LLM 不计算工业数值、不直接写业务数据。

**Tech Stack:** Java 17、Spring Boot 3.5、MyBatis-Plus、Sa-Token、dynamic-datasource、LangChain4j 1.18、Vue 3、Element Plus、Pinia、uni-app、Vitest、Maven、MySQL 8、SSE。

```java
package com.nexus.common.ai.decision.enums;

public enum DecisionRiskLevel {
    LOW,
    MEDIUM,
    HIGH,
    CRITICAL;

    public static DecisionRiskLevel fromScore(double score) {
        if (!Double.isFinite(score) || score < 0 || score > 100) {
            throw new IllegalArgumentException("风险分数必须在0到100之间");
        }
        if (score >= 70) return CRITICAL;
        if (score >= 40) return HIGH;
        if (score >= 20) return MEDIUM;
        return LOW;
    }
}
```

```java
@Component
@RequiredArgsConstructor
public class PrdScheduleInputAssembler {

    public ScheduleModel.ScheduleInput assemble(PrdSchedulePlanBo bo) {
        LocalDate startDate = toLocalDate(bo.getStartDate());
        int horizonDays = bo.getHorizonDays() == null ? 14 : Math.max(1, bo.getHorizonDays());
        LocalDate horizonEnd = startDate.plusDays(horizonDays - 1L);
        ScheduleModel.Strategy strategy = ScheduleModel.Strategy.parse(bo.getStrategy());

        List<PrdWorkOrder> workOrders = loadSchedulableWorkOrders(bo.getWorkshopId());
        if (workOrders.isEmpty()) {
            throw new ServiceException("没有可排产的工单(需存在 CREATED/APPROVED/RELEASED/IN_PROGRESS 状态工单)");
        }

        Map<Long, List<ScheduleModel.StepInfo>> routingMap = new HashMap<>();
        for (PrdWorkOrder workOrder : workOrders) {
            routingMap.computeIfAbsent(workOrder.getMaterialId(), this::loadRoutingSteps);
        }

        Map<LocalDate, Map<String, BigDecimal>> capacity = loadCapacity(startDate, horizonEnd);
        applyLaborCapacity(capacity, startDate, horizonEnd);

        ScheduleModel.ScheduleInput input = new ScheduleModel.ScheduleInput();
        input.setStartDate(startDate);
        input.setHorizonDays(horizonDays);
        input.setStrategy(strategy);
        input.setWorkOrders(workOrders.stream().map(this::toWoInfo).toList());
        input.setRoutingProvider(routingMap::get);
        input.setCapacityByDate(capacity);
        input.setHolidays(loadHolidays(startDate, horizonEnd));
        return input;
    }
}
```

```java
@Component
public class PrdShortageScenarioService {
    public PrdShortageResult simulate(PrdShortageInput base,
                                      List<PurchaseArrivalLine> purchaseLines,
                                      Map<Long, LocalDate> arrivalOverrides,
                                      LocalDate today) {
        if (!arrivalOverrides.isEmpty() && purchaseLines.isEmpty()) {
            throw new IllegalArgumentException("采购行到货覆盖缺少原始采购行数据");
        }
        PrdShortageInput copy = copyStockAndDemands(base);

            for (PurchaseArrivalLine line : purchaseLines) {
                LocalDate date = arrivalOverrides.getOrDefault(line.purchaseLineId(), line.originalDate());
                copy.addArrival(date, line.remainingQty());
            }
        }
        return PrdShortageSimulator.simulate(copy, today);
    }
}
```

```ts
export function riskTagType(level?: string): 'danger' | 'warning' | 'success' | 'info' {
  if (level === 'CRITICAL' || level === 'HIGH') return 'danger';
  if (level === 'MEDIUM') return 'warning';
  if (level === 'LOW') return 'success';
  return 'info';
}

export function formatDelta(value: number | null | undefined, unit = ''): string {
  if (value == null || Number.isNaN(value)) return '—';
  if (value < 0) return `改善 ${Math.abs(value)}${unit ? ` ${unit}` : ''}`;
  if (value > 0) return `增加 ${value}${unit ? ` ${unit}` : ''}`;
  return `无变化${unit ? ` ${unit}` : ''}`;
}

export function canApproveRun(run: { status?: string; stale?: boolean }): boolean {
  return run.status === 'WAITING_APPROVAL' && !run.stale;
}

export function sortScenarios<T extends { rank?: number; scenarioCode: string }>(items: T[]): T[] {
  return [...items].sort((a, b) => (a.rank ?? 999) - (b.rank ?? 999) || a.scenarioCode.localeCompare(b.scenarioCode));
}
```

## 4. 阶段验收门

### CP1：接口与表结构冻结

- 三表字段和索引通过合同测试；
- Port 不引用 ERP/Flow 类型；
- 状态、风险、动作枚举冻结。

### CP3：确定性分析冻结

- 同一快照重复运行得到相同评分和排名；
- 所有数值证据有稳定 evidenceKey；
- 不调用 LLM 也能完成风险、方案和动作规则计算。

### CP4：分析闭环

- 发起 API 返回 runId；
- Flow 时间线包含 Context/Shortage/APS/Rank/Guard/Action；
- 模型失败时仍进入 WAITING_APPROVAL；
- Flow 节点失败会把 run 标记 FAILED。

### CP5：后端 API 冻结

- PC/App 所需字段不再改名；
- SSE 所有 Decision 事件包含 string runId 和 seq；
- 刷新后 REST 可恢复全部状态。

### CP7：双端闭环

- PC/App 都能批准部分动作；
- STALE 状态不可批准；
- 创建的 ERP 对象保持草稿/未发布；
- 重复点击不会创建重复对象。
