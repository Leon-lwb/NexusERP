# NexusAI 工业交付风险闭环决策 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在不破坏现有通用 AI 对话链的前提下，交付“交付风险识别—缺料与 APS 推演—证据化决策—人工审批—安全执行—结果评测”的竞赛版工业决策闭环。

**Architecture:** 共享契约和持久化模型放在 `ruoyi-common-ai`；`ruoyi-erp` 通过 `IndustrialDecisionDataPort` 提供单一业务口径；`ruoyi-ai` 负责评分、证据、摘要、策略、状态机与 API；`ruoyi-flow` 通过 `DecisionWorkflowPort` 编排分析/执行流程。PC 与 App 只消费结构化 `DecisionResult`，LLM 不计算工业数值、不直接写业务数据。

**Tech Stack:** Java 17、Spring Boot 3.5、MyBatis-Plus、Sa-Token、dynamic-datasource、LangChain4j 1.18、Vue 3、Element Plus、Pinia、uni-app、Vitest、Maven、MySQL 8、SSE。

```java
package com.nexus.common.ai.decision.enums;

public enum DecisionRiskLevel {
    LOW,
    MEDIUM,
    HIGH,
    CRITICAL;

    public static DecisionRiskLevel fromScore(double score) {
        if (!Double.isFinite(score) || score < 0 || score > 100) {
            throw new IllegalArgumentException("风险分数必须在0到100之间");
        }
        if (score >= 70) return CRITICAL;
        if (score >= 40) return HIGH;
        if (score >= 20) return MEDIUM;
        return LOW;
    }
}
```

```ts
export function riskTagType(level?: string): 'danger' | 'warning' | 'success' | 'info' {
  if (level === 'CRITICAL' || level === 'HIGH') return 'danger';
  if (level === 'MEDIUM') return 'warning';
  if (level === 'LOW') return 'success';
  return 'info';
}

export function formatDelta(value: number | null | undefined, unit = ''): string {
  if (value == null || Number.isNaN(value)) return '—';
  if (value < 0) return `改善 ${Math.abs(value)}${unit ? ` ${unit}` : ''}`;
  if (value > 0) return `增加 ${value}${unit ? ` ${unit}` : ''}`;
  return `无变化${unit ? ` ${unit}` : ''}`;
}

export function canApproveRun(run: { status?: string; stale?: boolean }): boolean {
  return run.status === 'WAITING_APPROVAL' && !run.stale;
}

export function sortScenarios<T extends { rank?: number; scenarioCode: string }>(items: T[]): T[] {
  return [...items].sort((a, b) => (a.rank ?? 999) - (b.rank ?? 999) || a.scenarioCode.localeCompare(b.scenarioCode));
}
```
